\begin{frame}{Matrices}

Sea \(k\) un cuerpo. Una \alert{matriz de tamaño \(m\times n\)}
con coeficientes en \(k\) es una disposición rectangular de números:

\begin{center}
\begin{tikzpicture}[
    every node/.style={font=\small},
    >=stealth
]

% Matriz esquemática
\matrix (A) [
    matrix of math nodes,
    ampersand replacement=\&,
    left delimiter={(},
    right delimiter={)},
    row sep=2mm,
    column sep=2mm
] {
 a_{11} \& \cdots \& a_{1j} \& \cdots \& a_{1n} \\
 \vdots \&        \& \vdots \&        \& \vdots \\
 a_{i1} \& \cdots \& a_{ij} \& \cdots \& a_{in} \\
 \vdots \&        \& \vdots \&        \& \vdots \\
 a_{m1} \& \cdots \& a_{mj} \& \cdots \& a_{mn} \\
};

% Resaltar fila i
\draw[rounded corners]
  ([xshift=-2mm,yshift=2mm]A-3-1.north west)
  rectangle
  ([xshift=2mm,yshift=-2mm]A-3-5.south east);
\node[left=8mm of A-3-1] (fila) {\alert{fila \(i\)}};
\draw[->] (fila) -- (A-3-1.west);

% Resaltar columna j
\draw[rounded corners]
  ([xshift=-2mm,yshift=2mm]A-1-3.north west)
  rectangle
  ([xshift=2mm,yshift=-2mm]A-5-3.south east);
\node[above=5mm of A-1-3] (columna) {\alert{columna \(j\)}};
\draw[->] (columna) -- (A-1-3.north);

% Entrada (i,j)
\node[below right=10mm and 35mm of A-3-3] (entrada)
  {\alert{\(\displaystyle a_{ij}\): entrada \((i,j)\)}};

\draw[->]
  (entrada.west)
  to[bend left=20]
  (A-3-3.south east);

% Tamaño
\node[below=7mm of A] (hola) {
  \alert{\(m\) filas \qquad \(n\) columnas}
};

\end{tikzpicture}
\end{center}

\end{frame}
