\begin{frame}{Matrices}

Sea \(k\) un cuerpo. Una \alert{matriz de tamaño \(m\times n\)}
con coeficientes en \(k\) es una disposición rectangular de números:

\begin{frame}{Notación para matrices}

Las matrices se denotarán habitualmente por letras mayúsculas
\[
A,\ B,\ C,\ldots
\]

Si \(A\) es una matriz de \(m\) filas y \(n\) columnas, escribiremos
\[
A=(a_{ij}),
\qquad
1\leq i\leq m,\quad 1\leq j\leq n,
\]
donde \(a_{ij}\) es la entrada situada en la
\alert{fila \(i\)} y la \alert{columna \(j\)}.

\pause

El conjunto de todas las matrices de tamaño \(m\times n\)
con entradas en el cuerpo \(k\) se denota por
\[
\alert{M_{m\times n}(k)}.
\]

Para las matrices cuadradas escribiremos
\[
\alert{M_n(k)=M_{n\times n}(k)}.
\]

\end{frame}

\begin{frame}{Operaciones con matrices}

Sean
\[
A=(a_{ij}),\ B=(b_{ij})\in M_{m\times n}(k),
\qquad
\lambda\in k.
\]

\begin{block}{Suma}
La suma de \(A\) y \(B\) se realiza entrada a entrada:
\[
A+B=(a_{ij}+b_{ij}).
\]
\end{block}

\begin{block}{Producto por un escalar}
El producto de \(A\) por \(\lambda\) se obtiene multiplicando
cada entrada por \(\lambda\):
\[
\lambda A=(\lambda a_{ij}).
\]
\end{block}

\end{frame}

\begin{frame}{Operaciones con matrices}

Por ejemplo,
\[
\begin{pmatrix}
1&2&-1\\
0&3&2
\end{pmatrix}
+
\begin{pmatrix}
2&-1&4\\
1&0&-2
\end{pmatrix}
=
\begin{pmatrix}
3&1&3\\
1&3&0
\end{pmatrix}.
\]

\medskip

\alert{Para sumar dos matrices deben tener el mismo tamaño.}

\bigskip

Un ejemplo de producto por un escalar es:
\[
-2
\begin{pmatrix}
1&2&-1\\
0&3&2
\end{pmatrix}
=
\begin{pmatrix}
-2&-4&2\\
0&-6&-4
\end{pmatrix}.
\]

\end{frame}

\begin{frame}{Propiedades de las operaciones con matrices}

Para matrices de tamaños adecuados y escalares
\(\lambda,\mu\in k\):

\begin{align*}
A+(B+C) &= (A+B)+C,\\
A+B &= B+A,\\
A+0 &= A,\\
A+(-A) &= 0,
\end{align*}

\pause

\begin{align*}
A(B+C) &= AB+AC,\\
(A+B)C &= AC+BC,\\
A(BC) &= (AB)C,\\
\lambda(AB) &= (\lambda A)B=A(\lambda B).
\end{align*}

\pause

Si \(I_n\) denota la matriz identidad de tamaño \(n\), entonces
\[
I_mA=A=AI_n,
\qquad A\in M_{m\times n}(k).
\]

\medskip

\begin{alertblock}{Atención}
El producto de matrices no es, en general, conmutativo:
\[
AB\neq BA.
\]
Incluso puede ocurrir que \(AB\) esté definido y \(BA\) no.
\end{alertblock}

\end{frame}
